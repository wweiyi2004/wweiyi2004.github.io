cpp
#include <cstdio>

const int MAXV = 1000000000;
const int MN = 2000005;

int N, A[MN], C;

bool Check(int x) {
	for (int i = C, j = C; i >= 2; --i, ++j) {
		if ((A[i] >= x) == (A[i - 1] >= x))
			return A[i] >= x;
		if ((A[j] >= x) == (A[j + 1] >= x))
			return A[j] >= x;
	}
	return A[1] >= x;
}

int main() {
	scanf("%d", &N), C = (N + 1) / 2;
	for (int i = 1; i <= N; ++i) scanf("%d", &A[i]);
	int lb = 0, rb = MAXV, ans = -1;
	while (lb <= rb) {
		int mid = (lb + rb) >> 1;
		if (Check(mid)) ans = mid, lb = mid + 1;
		else rb = mid - 1;
	}
	printf("%d\n", ans);
	return 0;
}

思路

目标是在树上选一个最大点集 SSS ，使得 SSS 的导出子图由一些大小大于等于 222 的联通块构成 ，且权值和小于等于 eie_iei​ 。

考虑用树形dp解决这一问题，

  • 设 fx,i,0f_{x,i,0}fx,i,0​ 表示 xxx 子树，选了合法的 iii 个点， xxx 点没被选中的最小代价

转移就是考虑儿子 vvv 有没有被选中，这里还要考虑在 xxx 处把 vvv 改成选中的情况（若 vvv 没有选中任何儿子，这个子状态本来是非法的）。

求完每个案后最后一起背包到 ggg 中，注意 ggg 要从后往前取 minminmin ，考虑选择点个数为 333 的方案有可能劣于选择点个数为 444 的方案（这里感谢粉兔的大样例）。

#include <cstdio>
#include <algorithm>
#include <vector>

typedef long long LL;
const LL Inf = 0x3f3f3f3f3f3f3f3f;
const int MN = 2005;

int N, M, D[MN];
std::vector<int> G[MN];

inline void chkmin(LL &x, LL y) { x = x < y ? x : y; }
void Merge(LL (*fu)[3], int &n, LL (*fv)[3], int m) {
	static LL g[MN][3];
	for (int i = 0; i <= n + m; ++i)
		for (int p = 0; p < 3; ++p)
			g[i][p] = Inf;
	for (int i = 0; i <= n; ++i)
		for (int j = 0; j <= m; ++j) {
			LL mn02 = std::min(fv[j][0], fv[j][2]);
			chkmin(g[i + j][0], fu[i][0] + std::min(mn02, fv[j][1]));
			chkmin(g[i + j][1], fu[i][1] + fv[j][0]);
			chkmin(g[i + j + 1][2], fu[i][1] + fv[j][2]);
			chkmin(g[i + j + 1][2], fu[i][2] + fv[j][1]);
			chkmin(g[i + j + 2][2], fu[i][1] + fv[j][1]);
			chkmin(g[i + j][2], fu[i][2] + mn02);
		}
	for (int i = 0; i <= n + m; ++i)
		for (int p = 0; p < 3; ++p)
			fu[i][p] = g[i][p];
	n += m;
}

bool vis[MN];
int siz[MN];
LL f[MN][MN][3];
void DFS(int u, int p) {
	vis[u] = true;
	LL (*fu)[3] = f[u];
	siz[u] = 1;
	fu[0][0] = 0;
	fu[0][1] = D[u];
	fu[0][2] = fu[1][0] = fu[1][1] = fu[1][2] = Inf;
	for (int v : G[u]) if (v != p) {
		DFS(v, u);
		Merge(fu, siz[u], f[v], siz[v]);
	}
}

void Solve() {
	scanf("%d%d", &N, &M);
	for (int i = 1; i <= N; ++i) {
		scanf("%d", &D[i]);
		G[i].clear();
		vis[i] = false;
	}
	for (int i = 1; i <= M; ++i) {
		int x, y;
		scanf("%d%d", &x, &y);
		G[x].push_back(y);
		G[y].push_back(x);
	}
	static LL ans[MN];
	ans[0] = 0;
	for (int i = 1; i <= N; ++i) ans[i] = Inf;
	for (int i = 1; i <= N; ++i) if (!vis[i]) {
		DFS(i, 0);
		static LL tmp[MN];
		tmp[siz[i] + 1] = Inf;

			tmp[k] = std::min({f[i][k][0], f[i][k][1], f[i][k][2], tmp[k + 1]});
		for (int j = N; j >= 0; --j)
			for (int k = 0; k <= siz[i] && k <= j; ++k)
				chkmin(ans[j], ans[j - k] + tmp[k]);
	}
	int Q;
	scanf("%d", &Q);
	for (int q = 1; q <= Q; ++q) {
		int x;
		scanf("%d", &x);
		printf("%d%c", (int)(std::upper_bound(ans, ans + N + 1, x) - ans - 1), " \n"[q == Q]);
	}
}

int main() {
	int Tests;
	scanf("%d", &Tests);
	while (Tests--)
		Solve();
	return 0;
}

#include <cstdio>
#include <algorithm>

typedef unsigned long long i64;

i64 qPow(i64 b, i64 e) {
	i64 a = 1;
	for (; e; e >>= 1, b *= b)
		if (e & 1) a *= b;
	return a;
}
i64 gInv(i64 a) {
	i64 x = 1;
	x *= 2 - a * x;
	x *= 2 - a * x;
	x *= 2 - a * x;
	x *= 2 - a * x;
	x *= 2 - a * x;
	x *= 2 - a * x;
	return x;
}

int ctz(i64 a) {
	// deBruijn
	static const int table[64] = {
		0, 1, 2, 7, 3, 13, 8, 19,
		4, 25, 14, 28, 9, 34, 20, 40,
		5, 17, 26, 38, 15, 46, 29, 48,
		10, 31, 35, 54, 21, 50, 41, 57,
		63, 6, 12, 18, 24, 27, 33, 39,
		16, 37, 45, 47, 30, 53, 49, 56,
		62, 11, 23, 32, 36, 44, 52, 55,
		61, 22, 43, 51, 60, 42, 59, 58
	};
	static const i64 magic = 0x218A392CD3D5DBF;
	return table[((a & -a) * magic) >> 58];
}
struct mul {
	i64 v;
	int c;
	mul(i64 x) : v(x), c(0) {}
	operator i64() {
		if (c < 0 || c >= 64)
			return 0llu;
		return v << c;
	}
	mul &operator *= (i64 x) {
		int j = ctz(x);
		v *= x >> j;
		c += j;
		return *this;
	}
	mul &operator /= (i64 x) {
		int j = ctz(x);
		v *= gInv(x >> j);
		c -= j;
		return *this;
	}
};
mul Binom(i64 n, int m) {
	mul v(1);
	for (int i = 0; i < m; ++i)
		(v *= n - i) /= i + 1;
	return v;
}

const int MK = 250005;

int K;
i64 N, M, B[MK];

void Solve() {
	scanf("%d%llu%llu", &K, &N, &M), N -= K;
	for (int i = 1; i <= K; ++i) scanf("%llu", &B[i]);
	if (M == 1)
		return puts(N ? "0" : "1"), void();
	int c1 = 2, c2 = 0;
	for (int i = 2; i <= K; ++i)
		++(B[i] == B[i - 1] ? c1 : c2);
	i64 b1 = M - 1, b2 = M - 2;
	if (!c2) {
		printf("%llu\n", qPow(b1, N) * Binom(N + c1 - 1, c1 - 1));
		return ;
	}
	--c1, --c2;
	if (b1 & 1)
		std::swap(b1, b2),
		std::swap(c1, c2);
	i64 ans = 0;
	i64 pw1 = 1, pw2 = qPow(b2, N), ib2 = gInv(b2);
	mul binom1(1), binom2 = Binom(N + c2, c2);
	for (i64 i = 0; i <= N; ++i) {
		if (!pw1) break;
		ans += pw1 * pw2 * binom1 * binom2;
		if (i < N) {
			pw1 *= b1;
			pw2 *= ib2;
			(binom1 *= i + c1 + 1) /= i + 1;
			(binom2 /= N - i + c2) *= N - i;
		}
	}
	printf("%llu\n", ans);
}

int main() {

	scanf("%d", &Tests);
	while (Tests--) Solve();
	return 0;
}
