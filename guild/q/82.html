#include <cstdio>
#include <algorithm>
#include <limits>

typedef long long LL;
const int MN = 50005;

int N;
int ax[MN], ay[MN], bx[MN], by[MN], per[MN];

double Intersect(int i, int j) {
	LL p = (LL)(ax[i] * by[i] - ay[i] * bx[i]) * (ay[j] - by[j]) - (LL)(ax[j] * by[j] - ay[j] * bx[j]) * (ay[i] - by[i]);
	int q = (ax[i] - bx[i]) * (ay[j] - by[j]) - (ax[j] - bx[j]) * (ay[i] - by[i]);
	return (double)p / q;
}

bool cmp(int i, int j) {
	return (bx[i] - ax[i]) * (by[j] - ay[j]) < (bx[j] - ax[j]) * (by[i] - ay[i]);
}

int main() {
	scanf("%d", &N);
	for (int i = 1; i <= N; ++i) {
		scanf("%d%d%d%d", &ax[i], &ay[i], &bx[i], &by[i]);
		if (ay[i] > by[i])
			std::swap(ax[i], bx[i]),
			std::swap(ay[i], by[i]);
		per[i] = i;
	}
	std::sort(per + 1, per + N + 1, cmp);
	double Ans = std::numeric_limits<double>::lowest();
	for (int i = 2; i <= N; ++i)
		Ans = std::max(Ans, Intersect(per[i - 1], per[i]));
	printf("%.8f\n", Ans);
	return 0;
}

void DFS(int u, int f) {
	vis[u] = 1, dep[u] = dep[f] + 1;
	for (auto v : G[u]) {
		if (v == f) continue;
		if (vis[v]) {
			if (dep[v] < dep[u])
				col[u] ^= 1,
				printf("%d %d\n", u, v);
		}
		else DFS(v, u);
	}
	if (col[u]) printf("%d %d\n", u, f);
	else if (f) col[f] ^= 1, printf("%d %d\n", f, u);
}

		if (e & 1) a = (LL)a * b % Mod;
	return a;
}

	for (int i = 1; i <= N; ++i) {
		int x;
		scanf("%d", &x);
		for (int j = 1; j <= K; ++j) {
			f[j] = (f[j] + f[j - 1]) % Mod;
			g[j] = (g[j] + (LL)x * g[j - 1]) % Mod;
		}
	}
	printf("%lld\n", (LL)g[K] * qPow(f[K], Mod - 2) % Mod);
	return 0;
}

typedef long long LL;

LL N;

struct Matrix {
	int A[4][4];
	Matrix(int x = 0) {
		for (int i = 0; i < 4; ++i)
			for (int j = 0; j < 4; ++j)
				A[i][j] = i == j ? x : 0;
	}
};
Matrix operator * (const Matrix &p, const Matrix &q) {
	Matrix res;
	for (int i = 0; i < 4; ++i)
		for (int k = 0; k < 4; ++k)
			for (int j = 0; j < 4; ++j)
				res.A[i][j] = (res.A[i][j] + (LL)p.A[i][k] * q.A[k][j]) % Mod;
	return res;
}
Matrix qPow(Matrix b, LL e) {
	Matrix a(1);
	for (; e; e >>= 1, b = b * b)
		if (e & 1) a = a * b;
	return a;
}
